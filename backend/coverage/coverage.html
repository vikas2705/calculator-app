// Command server runs the calculator API and serves the built frontend.
//
// Configuration (environment variables):
//
//        PORT        port to listen on (default 8080)
//        STATIC_DIR  directory containing the built frontend (default ./static)
package main

import (
        "context"
        "errors"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculator/internal/api"
)

func main() {
        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        slog.SetDefault(logger)

        // Stop cleanly on Ctrl+C or `docker stop`.
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        if err := run(ctx, os.Getenv); err != nil {
                slog.Error("server stopped", "error", err)
                stop()
                os.Exit(1)
        }
}

// run serves until ctx is cancelled, then shuts down gracefully. getenv
// reads the configuration (os.Getenv in production, a map in tests).
func run(ctx context.Context, getenv func(string) string) error {
        port := valueOr(getenv("PORT"), "8080")
        staticDir := valueOr(getenv("STATIC_DIR"), "./static")

        if _, err := os.Stat(staticDir); err != nil {
                slog.Warn("static directory not found; only the API will be served", "dir", staticDir)
        }

        srv := &http.Server{
                Addr:              ":" + port,
                Handler:           api.NewRouter(staticDir),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        serveErr := make(chan error, 1)
        go func() {
                slog.Info("listening", "addr", srv.Addr, "static_dir", staticDir)
                serveErr <- srv.ListenAndServe()
        }()

        select {
        case err := <-serveErr:
                return err // failed to start, e.g. port already in use
        case <-ctx.Done():
        }

        slog.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-serveErr; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

func valueOr(value, fallback string) string {
        if value != "" {
                return value
        }
        return fallback
}

package api

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "strings"

        "calculator/internal/calculator"
)

// maxBodyBytes caps request bodies. A valid request is well under 200 bytes.
const maxBodyBytes = 1024

type calculateRequest struct {
        Operation string `json:"operation"`
        // Pointers let us tell an explicit null apart from 0. A plain float64
        // would silently turn [5, null] into [5, 0].
        Operands []*float64 `json:"operands"`
}

type calculateResponse struct {
        Result float64 `json:"result"`
}

// apiError is a client-facing failure: HTTP status, stable code, readable message.
type apiError struct {
        status  int
        code    ErrorCode
        message string
}

// domainErrors maps calculator errors to API errors.
var domainErrors = map[error]apiError{
        calculator.ErrDivisionByZero:   {http.StatusUnprocessableEntity, CodeDivisionByZero, "Cannot divide by zero."},
        calculator.ErrNegativeSqrt:     {http.StatusUnprocessableEntity, CodeNegativeSqrt, "Cannot take the square root of a negative number."},
        calculator.ErrUndefinedResult:  {http.StatusUnprocessableEntity, CodeUndefinedResult, "The result is undefined."},
        calculator.ErrOverflow:         {http.StatusUnprocessableEntity, CodeOverflow, "The result is too large."},
        calculator.ErrUnknownOperation: {http.StatusBadRequest, CodeInvalidOperation, "Unknown operation."},
        calculator.ErrOperandCount:     {http.StatusBadRequest, CodeInvalidOperands, "Wrong number of operands."},
        calculator.ErrNonFiniteOperand: {http.StatusBadRequest, CodeInvalidOperands, "Operands must be finite numbers."},
}

func handleCalculate(w http.ResponseWriter, r *http.Request) {
        var req calculateRequest
        if err := decodeJSON(w, r, &req); err != nil {
                writeAPIError(w, *err)
                return
        }

        op := calculator.Operation(req.Operation)
        operands, err := validateOperands(op, req.Operands)
        if err != nil {
                writeAPIError(w, *err)
                return
        }

        result, calcErr := calculator.Calculate(op, operands)
        if calcErr != nil {
                mapped, ok := domainErrors[calcErr]
                if !ok {
                        slog.Error("unmapped calculator error", "error", calcErr)
                        mapped = apiError{http.StatusInternalServerError, CodeInternal, "Something went wrong."}
                }
                writeAPIError(w, mapped)
                return
        }

        writeJSON(w, http.StatusOK, calculateResponse{Result: result})
}

// decodeJSON reads exactly one JSON object from the body into dst.
func decodeJSON(w http.ResponseWriter, r *http.Request, dst any) *apiError {
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()

        err := dec.Decode(dst)
        if err == nil {
                // Anything after the first value (another object, stray text) is rejected.
                if extra := dec.Decode(&struct{}{}); !errors.Is(extra, io.EOF) {
                        err = extra
                        if err == nil {
                                err = errors.New("trailing data")
                        }
                }
        }

        var tooLarge *http.MaxBytesError
        switch {
        case err == nil:
                return nil
        case errors.As(err, &tooLarge):
                return &apiError{http.StatusRequestEntityTooLarge, CodePayloadTooLarge,
                        fmt.Sprintf("Request body must not exceed %d bytes.", maxBodyBytes)}
        case errors.Is(err, io.EOF):
                return &apiError{http.StatusBadRequest, CodeInvalidJSON, "Request body is empty."}
        default:
                return &apiError{http.StatusBadRequest, CodeInvalidJSON,
                        `Request body must be a single JSON object, e.g. {"operation":"add","operands":[1,2]}.`}
        }
}

func validateOperands(op calculator.Operation, raw []*float64) ([]float64, *apiError) {
        arity, ok := op.Arity()
        if !ok {
                return nil, &apiError{http.StatusBadRequest, CodeInvalidOperation,
                        fmt.Sprintf(`"operation" must be one of: %s.`, operationList())}
        }
        if len(raw) != arity {
                return nil, &apiError{http.StatusBadRequest, CodeInvalidOperands,
                        fmt.Sprintf(`"%s" needs exactly %d operand(s) in "operands".`, op, arity)}
        }

        operands := make([]float64, len(raw))
        for i, x := range raw {
                if x == nil {
                        return nil, &apiError{http.StatusBadRequest, CodeInvalidOperands, `"operands" must not contain null.`}
                }
                operands[i] = *x
        }
        return operands, nil
}

func operationList() string {
        ops := calculator.Operations()
        names := make([]string, len(ops))
        for i, op := range ops {
                names[i] = string(op)
        }
        return strings.Join(names, ", ")
}

func writeAPIError(w http.ResponseWriter, e apiError) {
        writeError(w, e.status, e.code, e.message)
}

// withRequestLogging writes one structured log line per request.
func withRequestLogging(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}

                next.ServeHTTP(rec, r)

                slog.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration", time.Since(start).Round(time.Microsecond),
                )
        })
}

// statusRecorder captures the status code written by the wrapped handler.
type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (r *statusRecorder) WriteHeader(status int) {
        r.status = status
        r.ResponseWriter.WriteHeader(status)
}

// ErrorCode is a stable, machine-readable error identifier returned to clients.
type ErrorCode string

const (
        CodeInvalidJSON      ErrorCode = "INVALID_JSON"
        CodeInvalidOperation ErrorCode = "INVALID_OPERATION"
        CodeInvalidOperands  ErrorCode = "INVALID_OPERANDS"
        CodePayloadTooLarge  ErrorCode = "PAYLOAD_TOO_LARGE"
        CodeDivisionByZero   ErrorCode = "DIVISION_BY_ZERO"
        CodeNegativeSqrt     ErrorCode = "NEGATIVE_SQRT"
        CodeUndefinedResult  ErrorCode = "UNDEFINED_RESULT"
        CodeOverflow         ErrorCode = "OVERFLOW"
        CodeMethodNotAllowed ErrorCode = "METHOD_NOT_ALLOWED"
        CodeNotFound         ErrorCode = "NOT_FOUND"
        CodeInternal         ErrorCode = "INTERNAL_ERROR"
)

// ErrorCodes lists every code the API can return. The OpenAPI spec is tested against it.
var ErrorCodes = []ErrorCode{
        CodeInvalidJSON, CodeInvalidOperation, CodeInvalidOperands, CodePayloadTooLarge,
        CodeDivisionByZero, CodeNegativeSqrt, CodeUndefinedResult, CodeOverflow,
        CodeMethodNotAllowed, CodeNotFound, CodeInternal,
}

type errorResponse struct {
        Error errorDetail `json:"error"`
}

type errorDetail struct {
        Code    ErrorCode `json:"code"`
        Message string    `json:"message"`
}

// writeJSON encodes v before writing anything, so an encoding failure can
// still produce a clean 500 instead of a half-written response.
func writeJSON(w http.ResponseWriter, status int, v any) {
        body, err := json.Marshal(v)
        if err != nil {
                slog.Error("encoding response", "error", err)
                status = http.StatusInternalServerError
                body = []byte(`{"error":{"code":"INTERNAL_ERROR","message":"Something went wrong."}}`)
        }
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _, _ = w.Write(append(body, '\n'))
}

func writeError(w http.ResponseWriter, status int, code ErrorCode, message string) {
        writeJSON(w, status, errorResponse{Error: errorDetail{Code: code, Message: message}})
}

// NewRouter returns the application's HTTP handler.
// staticDir holds the built frontend. It is served at "/".
func NewRouter(staticDir string) http.Handler {
        mux := http.NewServeMux()

        mux.HandleFunc("POST /api/v1/calculate", handleCalculate)
        mux.HandleFunc("/api/v1/calculate", handleMethodNotAllowed(http.MethodPost))
        mux.HandleFunc("/api/", handleAPINotFound)

        mux.HandleFunc("GET /health", handleHealth)
        mux.Handle("GET /openapi.json", docs.SpecHandler())
        mux.Handle("GET /docs", docs.UIHandler())

        mux.Handle("/", http.FileServerFS(os.DirFS(staticDir)))

        return withRequestLogging(mux)
}

func handleHealth(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func handleMethodNotAllowed(allowed string) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Allow", allowed)
                writeError(w, http.StatusMethodNotAllowed, CodeMethodNotAllowed,
                        "Method "+r.Method+" is not allowed. Use "+allowed+".")
        }
}

func handleAPINotFound(w http.ResponseWriter, _ *http.Request) {
        writeError(w, http.StatusNotFound, CodeNotFound, "No such API endpoint. See /docs.")
}

// Package calculator implements the arithmetic operations offered by the API.
//
// It is pure: no I/O, no HTTP, no global state. Every function takes numbers
// and returns a number or an error, which makes it trivial to test.
package calculator

import (
        "math"
        "slices"
)

// Operation identifies a calculator operation. Its value is the name used in the API.
type Operation string

const (
        Add        Operation = "add"
        Subtract   Operation = "subtract"
        Multiply   Operation = "multiply"
        Divide     Operation = "divide"
        Power      Operation = "power"
        Sqrt       Operation = "sqrt"
        Percentage Operation = "percentage"
)

// definition describes how many operands an operation takes and how to compute it.
type definition struct {
        arity int
        apply func(x []float64) (float64, error)
}

var definitions = map[Operation]definition{
        Add:        {arity: 2, apply: func(x []float64) (float64, error) { return x[0] + x[1], nil }},
        Subtract:   {arity: 2, apply: func(x []float64) (float64, error) { return x[0] - x[1], nil }},
        Multiply:   {arity: 2, apply: func(x []float64) (float64, error) { return x[0] * x[1], nil }},
        Divide:     {arity: 2, apply: divide},
        Power:      {arity: 2, apply: power},
        Sqrt:       {arity: 1, apply: sqrt},
        Percentage: {arity: 1, apply: func(x []float64) (float64, error) { return x[0] / 100, nil }},
}

// Operations returns every supported operation, sorted by name.
func Operations() []Operation {
        ops := make([]Operation, 0, len(definitions))
        for op := range definitions {
                ops = append(ops, op)
        }
        slices.Sort(ops)
        return ops
}

// Arity returns how many operands op takes, and false if op is not supported.
func (op Operation) Arity() (int, bool) {
        def, ok := definitions[op]
        return def.arity, ok
}

// Calculate applies op to operands.
//
// Operands must be finite and match the operation's arity. The result is
// always finite: a result too large for float64 returns ErrOverflow, and a
// mathematically undefined result returns ErrUndefinedResult. Negative zero
// is returned as plain zero.
func Calculate(op Operation, operands []float64) (float64, error) {
        def, ok := definitions[op]
        if !ok {
                return 0, ErrUnknownOperation
        }
        if len(operands) != def.arity {
                return 0, ErrOperandCount
        }
        for _, x := range operands {
                if math.IsNaN(x) || math.IsInf(x, 0) {
                        return 0, ErrNonFiniteOperand
                }
        }

        result, err := def.apply(operands)
        if err != nil {
                return 0, err
        }
        return checkResult(result)
}

func divide(x []float64) (float64, error) {
        if x[1] == 0 {
                return 0, ErrDivisionByZero
        }
        return x[0] / x[1], nil
}

func power(x []float64) (float64, error) {
        base, exponent := x[0], x[1]
        // 0 raised to a negative power is 1/0. math.Pow would return +Inf,
        // which would otherwise be misreported as an overflow.
        if base == 0 && exponent < 0 {
                return 0, ErrDivisionByZero
        }
        return math.Pow(base, exponent), nil
}

func sqrt(x []float64) (float64, error) {
        if x[0] < 0 {
                return 0, ErrNegativeSqrt
        }
        return math.Sqrt(x[0]), nil
}

// checkResult turns non-finite results into errors and normalises -0 to 0.
func checkResult(r float64) (float64, error) {
        switch {
        case math.IsNaN(r):
                return 0, ErrUndefinedResult
        case math.IsInf(r, 0):
                return 0, ErrOverflow
        case r == 0:
                return 0, nil // also turns -0 into 0
        default:
                return r, nil
        }
}

// Package docs serves the OpenAPI specification and a Swagger UI page for it.
package docs

import (
        _ "embed"
        "net/http"
)

// Spec is the OpenAPI 3 document describing the API.
//
//go:embed openapi.json
var Spec []byte

//go:embed swagger.html
var swaggerPage []byte

// SpecHandler serves the OpenAPI document.
func SpecHandler() http.Handler {
        return staticBytes("application/json; charset=utf-8", Spec)
}

// UIHandler serves the Swagger UI page, which loads the spec from /openapi.json.
func UIHandler() http.Handler {
        return staticBytes("text/html; charset=utf-8", swaggerPage)
}

func staticBytes(contentType string, body []byte) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
                w.Header().Set("Content-Type", contentType)
                _, _ = w.Write(body)
        })
}
